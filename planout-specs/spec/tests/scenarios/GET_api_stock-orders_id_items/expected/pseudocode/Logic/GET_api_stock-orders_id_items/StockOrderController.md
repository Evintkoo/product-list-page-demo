/**
 * GET /api/stock-orders/:id/items
 * Retrieves a paginated list of StockOrderItem records for a specific StockOrder.
 */
function getStockOrderItems(request, response) {
    const { id: stockOrderId } = request.params;
    const { page = 1, limit = 20, product_name_filter } = request.query;
    const companyId = request.user.company_id; // Derived from auth context

    // 1. Validate existence and ownership
    const stockOrder = database.StockOrder.findOne({
        where: { 
            id: stockOrderId,
            company_id: companyId 
        }
    });

    if (!stockOrder) {
        return response.status(404).json({ error: "Stock Order not found or access denied" });
    }

    // 2. Build query for items
    const queryOptions = {
        where: { stock_order_id: stockOrderId },
        order: [['created_at', 'ASC']] // Insertion order
    };

    if (product_name_filter) {
        queryOptions.where.product_name = { [Op.like]: `%${product_name_filter}%` };
    }

    // 3. Fetch paginated items
    const items = database.StockOrderItem.findMany(queryOptions, {
        offset: (page - 1) * limit,
        limit: limit
    });

    // 4. Construct response
    // Note: total_order_qty is retrieved from the parent StockOrder record
    // The "No" column is generated by the frontend/serializer as an index + 1
    return response.status(200).json({
        total_order_qty: stockOrder.total_order_qty,
        items: items.map((item, index) => ({
            ...item,
            row_number: ((page - 1) * limit) + index + 1
        }))
    });
}
